function boneShape(ctx,x,y,len,rot,a=1,trail=0){ctx.save();ctx.translate(x,y);ctx.rotate(rot);ctx.globalAlpha=a; if(trail){ctx.shadowColor='#f5e2bf';ctx.shadowBlur=trail;} ctx.fillStyle='#f8f0dd';ctx.strokeStyle='#8f7d63';ctx.lineWidth=1.2;ctx.beginPath();ctx.moveTo(-len*.45,-len*.10);ctx.lineTo(len*.45,-len*.10);ctx.lineTo(len*.45,len*.10);ctx.lineTo(-len*.45,len*.10);ctx.closePath();ctx.fill();ctx.stroke(); for(const sx of [-1,1])for(const oy of [-1,1]){ctx.beginPath();ctx.arc(sx*len*.49,oy*len*.14,len*.15,0,Math.PI*2);ctx.fill();ctx.stroke();} ctx.restore();}

 if(fx.kind==='bonemerang'){

  const dir=ret?1:-1;
  const x=L(ax,bx,out), y=L(ay,by,out)-Math.sin(Math.PI*out)*(ret?24:42);

    const hit=C((t-.34)/.34);

 }

  }

 }
}
